#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N=200005;
int n,a[N],sum[N];
stack<int> st;
signed main(){
	ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
	cin>>n;
	for(int i=1;i<=n;i++) cin>>a[i];
	for(int i=n;i>=1;i--){
		while(!st.empty()&&a[i]-(st.top()-i)>a[st.top()]) st.pop();
		if(st.empty()) sum[i]=a[i]*a[i];
		else sum[i]=sum[st.top()]+(a[i]-(st.top()-i)+a[i])*(st.top()-i);
		st.push(i);
	}
	for(int i=1;i<=n;i++) cout<<sum[i]<<'\n';
	return 0;
}
