async function login(){ let account=document.getElementById("account").value; let password=document.getElementById("password").value; let msg=document.getElementById("msg"); if(!account||!password){ msg.innerHTML="请输入账号和密码"; return; } let email=account; // 如果输入的不是邮箱，就按用户名查询 if(!account.includes("@")){ let {data,error}=await client .from("profile") .select("email") .eq("username",account) .single(); if(error||!data){ msg.innerHTML="用户名不存在"; return; } email=data.email; } // 使用邮箱登录 Supabase Auth let {data,error}=await client.auth.signInWithPassword({ email:email, password:password }); if(error){ msg.innerHTML="密码错误"; return; } msg.innerHTML="登录成功"; setTimeout(()=>{ location.href="index.html"; },800); }
